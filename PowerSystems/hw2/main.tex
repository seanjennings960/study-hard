\section{Formulas}
\subsection{Phasors}
$\bar{V}$ denotes the phasor of RMS value $V$ and phase angle
$\delta$ and represents the sinusoid:
\begin{equation}
    \bar{V} = V \angle \delta \Leftrightarrow
        \sqrt{2} V \cos(\omega t + \delta)
\end{equation}
where $\omega$ is the reference angular frequency, which
is assumed constant for any phasors within a given computation.

\subsection{Complex Power}

Complex power $S$ is related to voltage and current phasors,
$\bar{V}$ and $\bar{I}$, respectively by
\begin{equation}
    S = \bar{V} \bar{I}^*
    \label{eq:complex-power-from-phasors}
\end{equation}
where $I^*$ denote the complex conjugate. Its real component
is called complex power, and is denoted $P$; its imaginary
component is called reactive power, denoted $Q$.
\begin{equation}
    S = P + jQ
    \label{eq:complex-power-components}
\end{equation}
where $j = \sqrt{-1}$. Its magnitude, $|S|$ is called apparent
power.

If $Z$ is the impedance between a given $\bar{V}$ and $\bar{I}$,
such that $\bar{V} = Z \bar{I}$,
(\ref{eq:complex-power-from-phasors})
can be rewritten:
\begin{align}
    S &= \frac{V^2}{Z^*}
        \label{eq:complex-power-from-voltage} \\
        &= ZI^2
        \label{eq:complex-power-from-current}
\end{align}
Note: we have used the property $\bar{V}\bar{V}^* = V^2$
and the analogy for current.

\subsection{Instantaneous Power}
For voltage $v(t)$ and current $i(t)$, given by
\begin{equation}
    v(t) = V_{max} \cos(\omega t + \delta) \qquad
        i(t) = I_{max} \cos(\omega t + \beta),
\end{equation}

Instantaneous power, is given by $p(t) = v(t)i(t)$,
which via use of product-to-sum and sum/difference trig identities
reduces to
\begin{equation}
    p(t) = VI_R\biggl( 1 + \cos(2[\omega t + \delta])\biggr)
        + VI_X \sin(2[\omega t + \delta]).
    \label{eq:instantaneous-power-raw}
\end{equation}
$I_R$ is called the real current, given by
$I_R = I \cos(\delta - \beta)$
and $I_X$ is called the reactive current, given by
$I_X = I \sin(\delta - \beta)$. By noting the similarity
to Euler's formula, we have
\begin{equation}
    P = VI_R \qquad Q = VI_X
    \label{eq:instantaneous-to-complex-relation}
\end{equation}
So equation (\ref{eq:instantaneous-power}) can be rewritten:
\begin{equation}
    p(t) = P\biggl( 1 + \cos(2[\omega t + \delta])\biggr)
        + Q \sin(2[\omega t + \delta]).
    \label{eq:instantaneous-power}
\end{equation}

\subsection{Power Factor}
Power factor, $\pf$ is given by
\begin{align}
    \pf &= \cos(\delta - \beta) 
    \label{eq:power-factor} \\
        &= \cos(\angle(S))
    \label{eq:power-factor-complex}
\end{align}

\subsection{Sign Conventions} 
When current enters the terminal designated as having positive 
voltage, we use the "load convention" and positive power
means it is {\it absorbed}.
When current enters the negative terminal, we use the "generator
convention and positive power means it is {\it delivered}.
Table (\ref{table:sign-conventions}) summarizes the parlance.
\begin{table}[h]
    \centering
    \begin{tabular}{|c|c|c|}
        \hline
        Convention & Load & Generator \\
        \hline
        P $>$ 0 & real power absorbed & real power delivered \\
        P $<$ 0 & real power delivered & real power absorbed \\
        Q $>$ 0 & positive reactive power absorbed &
            positive reactive power delivered \\
        Q $<$ 0 & positive reactive power delivered &
            positive reactive power absorbed \\
        \hline
    \end{tabular}
    \caption{Summary of Sign Convention}
    \label{table:sign-conventions}
\end{table}

\subsection{Complex from Real Power}
Given a load which absorbed real power $P$ at a power factor
$\pf$, the complex power is given by:
\begin{equation}
    S = P + (P \cdot \tan(d \cdot \arccos(\pf)))
    \label{eq:complex-from-real}
\end{equation}
where $d = 1$ if the load is lagging, and $d = -1$
if it is leading.

\subsection{Power Factor correction}
The amount of reactive power $\Delta Q$ that needs to be
absorbed to correct a load with real power $P$ and power
factor $\pf$ to a new target power factor $\pf_t$, with
the real power comsumption constant is:
\begin{equation}
    \Delta Q = P (\tan (\arccos(\pf)) - \tan (\arccos(\pf_t)))
    \label{eq:reactive-power-correction}
\end{equation}

\subsection{Resistance/Reactance Equivalents in Parallel and Series}
If a load has impedance $Z$ can be represented by two equivalent
circuits. The first has a resistance $R_s$ in series with a
reactance $X_s$ so that it has total impedance $Z_s = R_s + j X_s$.
Thus,
\begin{equation}
    R_s = \text{real}(Z) \qquad X_s = \text{imag}(Z).
\end{equation}
where $\text{real}(Z)$ is the real component of $Z$ and
$\text{imag}(Z)$ is the imaginary components such that
$Z = \text{real}(Z) + j \text{imag}(Z)$.

The second equivalent circuit has a resistance $R_p$ in parallel
with an impedance $X_p$. The equations
\begin{equation}
    R_p = \frac{|Z|^2}{\text{real}(Z)} \qquad
    X_p = \frac{|Z|^2}{\text{imag}(Z)}
    \label{eq:parallel-equivalent-impedance}
\end{equation}
gives the resistance and reactance in the parallel circuit
as a function of the equivalent impedance, where $|Z|^2 = ZZ^*$
is the square of the magnitude of $Z$.

\begin{letterenum}
    \item Since $S = Z I^2$, the angle of impedance determines
        the how much power is real vs. reactive ($I$ is a magnitude!):
        \begin{equation*}
            P = I^2 \cos(\angle(Z)) \qquad Q = I^2 \sin(\angle(Z))
        \end{equation*}
    \item The voltage angle is $\angle(V) = \angle(Z) + \angle(I) = \angle(Z)$
    so power is:
    \begin{equation*}
        p(t) = \instpower{P}{Q}{\angle(Z)}
    \end{equation*}

    \item If you have more (less) capacitance than inductance
        (as compared by reactance), the reactive power
        angle of the impedance becomes negative (positive), and thus
        the reactive power becomes negative (positive) as well.

        The relative magnitude of resistance compared to the total
        reactance is also importance. More (less) resistance
        means the angle will be smaller (larger), and thus more
        (less) real power,

        When $\omega^2LC = 1$, we have the following expression for
        reactance
        \begin{equation*}
            X = \omega L - \frac{1}{\omega C}
                = \frac{\omega^2 LC - 1}{j \omega C}
        \end{equation*}
        So the impedance angle is 0 and the reactive power term
        disappears. The instantaneous power curve is then a 
        sinusoid with mean and amplitude of real power, so it
        has peaks of 0 and $2P$.
\end{letterenum}
